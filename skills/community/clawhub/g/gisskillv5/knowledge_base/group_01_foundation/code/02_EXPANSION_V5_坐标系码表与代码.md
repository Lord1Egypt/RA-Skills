### 2.2 GIS软件中快速计算中央子午线

```python
# Python: 根据经度计算3度带/6度带中央子午线
def central_meridian_3deg(lon):
    """3度带中央子午线"""
    zone = int((lon + 1.5) / 3)
    return zone * 3

def central_meridian_6deg(lon):
    """6度带中央子午线"""
    zone = int(lon / 6) + 1
    return zone * 6 - 3

def get_epsg_3deg(lon):
    """CGCS2000 3度带EPSG"""
    zone = int((lon + 1.5) / 3)
    return 4513 + (zone - 25)  # 25带→4513

def get_epsg_6deg(lon):
    """CGCS2000 6度带EPSG"""
    zone = int(lon / 6) + 1
    return 4480 + (zone - 13)  # 13带→4480

# 示例
lon = 116.4  # 北京
print(f"3度带: {central_meridian_3deg(lon)}°E, EPSG:{get_epsg_3deg(lon)}")
print(f"6度带: {central_meridian_6deg(lon)}°E, EPSG:{get_epsg_6deg(lon)}")
```

## 三、七参数/四参数/三参数坐标转换完整代码

### 3.1 七参数（Bursa-Wolf模型）

**适用场景**: 不同椭球基准间转换（如WGS84↔北京54），全国或大区域

**七参数**: ΔX, ΔY, ΔZ(平移), Rx, Ry, Rz(旋转,角秒), M(尺度,ppm)

```python
import numpy as np

def bursa_wolf_7param(x, y, z, dx, dy, dz, rx, ry, rz, m):
    """
    Bursa-Wolf 七参数转换（WGS84 → 目标坐标系）
    
    参数:
        x, y, z: 输入WGS84地心坐标
        dx, dy, dz: 平移参数(米)
        rx, ry, rz: 旋转参数(角秒)
        m: 尺度参数(ppm)
    """
    # 角秒转弧度
    rx_rad = rx / 3600.0 * np.pi / 180.0
    ry_rad = ry / 3600.0 * np.pi / 180.0
    rz_rad = rz / 3600.0 * np.pi / 180.0
    
    # ppm转尺度因子
    scale = 1.0 + m / 1000000.0
    
    # 旋转矩阵
    R = np.array([
        [1, -rz_rad, ry_rad],
        [rz_rad, 1, -rx_rad],
        [-ry_rad, rx_rad, 1]
    ])
    
    src = np.array([x, y, z])
    target = np.array([dx, dy, dz]) + scale * R @ src
    return target[0], target[1], target[2]

def wgs84_lla_to_ecef(lat, lon, h=0):
    """WGS84 大地坐标(经纬度) → 地心直角坐标"""
    a = 6378137.0  # WGS84长半轴
    f = 1 / 298.257223563
    e2 = 2*f - f*f
    
    lat_rad = np.radians(lat)
    lon_rad = np.radians(lon)
    
    N = a / np.sqrt(1 - e2 * np.sin(lat_rad)**2)
    
    x = (N + h) * np.cos(lat_rad) * np.cos(lon_rad)
    y = (N + h) * np.cos(lat_rad) * np.sin(lon_rad)
    z = (N * (1 - e2) + h) * np.sin(lat_rad)
    
    return x, y, z

def ecef_to_lla(x, y, z, a, f_inv):
    """地心直角坐标 → 大地坐标(通用椭球)"""
    f = 1.0 / f_inv
    e2 = 2*f - f*f
    
    lon = np.degrees(np.arctan2(y, x))
    
    p = np.sqrt(x**2 + y**2)
    lat = np.degrees(np.arctan2(z, p * (1 - e2)))
    
    for _ in range(5):  # 迭代5次
        N = a / np.sqrt(1 - e2 * np.sin(np.radians(lat))**2)
        h = p / np.cos(np.radians(lat)) - N
        lat = np.degrees(np.arctan2(z, p * (1 - e2 * N / (N + h))))
    
    return lat, lon, h

# === 使用示例 ===
# 已知区域七参数（示例值，实际需测绘局提供）
params = {
    'dx': -123.456, 'dy': 78.901, 'dz': -45.678,  # 米
    'rx': -1.234, 'ry': 2.345, 'rz': 3.456,        # 角秒
    'm': 1.567                                        # ppm
}

# WGS84坐标(北京)
lat, lon, h = 39.9042, 116.4074, 50.0
x, y, z = wgs84_lla_to_ecef(lat, lon, h)
target_x, target_y, target_z = bursa_wolf_7param(x, y, z, **params)
print(f"目标坐标系坐标: X={target_x:.3f}, Y={target_y:.3f}, Z={target_z:.3f}")
```

### 3.2 四参数（平面转换）

**适用场景**: 同一椭球基准不同投影间转换，或小范围（<100km）不同基准近似转换

**四参数**: Δx, Δy(平移), α(旋转角), k(尺度)

```python
def four_param_2d(x, y, dx, dy, alpha_deg, k):
    """
    四参数二维平面转换
    
    参数:
        x, y: 输入坐标
        dx, dy: 平移量
        alpha_deg: 旋转角(度)
        k: 尺度因子(如1.000001)
    """
    alpha = np.radians(alpha_deg)
    cos_a = np.cos(alpha)
    sin_a = np.sin(alpha)
    
    x_new = dx + k * (x * cos_a + y * sin_a)
    y_new = dy + k * (-x * sin_a + y * cos_a)
    
    return x_new, y_new

# 示例：地方独立坐标→CGCS2000高斯投影(成都区域 102°E)
params = {'dx': 500000.0, 'dy': 300000.0, 'alpha_deg': 0.0012, 'k': 0.999996}
x_local, y_local = 1000.0, 2000.0
x_cgcs, y_cgcs = four_param_2d(x_local, y_local, **params)
print(f"CGCS2000: X={x_cgcs:.3f}, Y={y_cgcs:.3f}")
```

### 4.2 GNSS大地高 → 1985正常高转换

```
核心公式: H_正常高 = H_大地高 - ζ_高程异常

高程异常ζ来源:
  1. EGM2008全球重力场模型(分辨率1′≈2km) — 免费
  2. CQG2000中国重力似大地水准面(分辨率2.5′≈5km) — 官方
  3. 省级精化似大地水准面(分辨率2′×2′≈3.7km，精度±2.5cm) — 省级测绘局
  4. RTK实测高程拟合(精度±2cm) — 需外业测量
```

```python
# GNSS大地高→正常高转换
def geoid_to_normal_height(ellipsoid_height, geoid_undulation):
    """
    大地高 → 正常高
    H_normal = H_ellipsoid - N_geoid
    
    参数:
        ellipsoid_height: GNSS测得的大地高(m)
        geoid_undulation: 高程异常值(m)，可通过EGM2008/CQG2000查询
    """
    return ellipsoid_height - geoid_undulation

# 示例：成都某点 WGS84大地高500m，EGM2008高程异常-42.3m
h_normal = geoid_to_normal_height(500.0, -42.3)
print(f"1985正常高: {h_normal:.3f}m")
```

### 4.3 ArcPy批量高程转换

```python
# ArcPy: CGCS2000大地高→1985正常高(需安装EGM2008附加包)
import arcpy

def batch_ellipsoid_to_orthometric(fc, geoid_raster):
    """批量大地高转正常高"""
    arcpy.ddd.AddSurfaceInformation(
        fc, geoid_raster, "Z_MIN;Z_MAX;Z_MEAN"
    )
    arcpy.management.CalculateField(
        fc, "NORMAL_H", "!Z_MEAN! - !GEOID_OFFSET!", "PYTHON3"
    )
```

### 5.2 地方坐标→国家坐标系转换策略

```
策略A: 有公共点(推荐)
  1. 在地方坐标系和国家坐标系中至少3个公共点
  2. 计算四参数/七参数
  3. 验证残差<限差(一般5cm)
  4. 批量转换

策略B: 无公共点但有转换参数
  1. 向当地测绘局/规划局申请转换参数
  2. 直接使用四参数/七参数转换

策略C: 无参数需自建(最后手段)
  1. 外业RTK采集至少3个控制点(地方+国家双坐标)
  2. 最小二乘平差计算转换参数
  3. 精度评定(内符合/外符合)
```
